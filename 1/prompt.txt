## The acceptance test (from the brief, Step 1)

```python
def test_every_entity_type_survives_reopen(tmp_path):
    db = str(tmp_path / "odi.db")
    backend = LocalBackend(db_path=db)

    import json, re

    def json_of(text: str) -> dict:
        return json.loads(re.search(r"({.*})", text).group(1))

    job_id = json_of(backend.call("add_job", {"name": "J", "description": "d"}))["id"]
    step_id = json_of(
        backend.call("add_step", {"job_id": job_id, "name": "S", "order": 1})
    )["id"]
    backend.call(
        "add_outcome",
        {
            "job_id": job_id,
            "statement": "Min x",
            "importance": 8.0,
            "satisfaction": 4.0,
            "step_id": step_id,
        },
    )
    backend.call(
        "add_customer_outcome",
        {
            "job_id": job_id,
            "customer_value": {
                "direction": "minimize",
                "unit_of_measure": "time",
                "object_of_control": "x",
                "statement": "Min x",
                "importance": 8.0,
                "satisfaction": 4.0,
                "value_metrics": [
                    {
                        "name": "m",
                        "direction": "minimize",
                        "unit_of_measure": "s",
                        "importance": 6.0,
                        "satisfaction": 2.0,
                    }
                ],
            },
            "business_value": {"category": "cost_reduction", "statement": "Save"},
        },
    )

    repo = SQLiteRepository(db_path=db)
    assert repo.get_job(job_id).name == "J"
    assert repo.get_step(step_id).order == 1
    outcome = repo.get_outcome(repo.get_outcomes(job_id)[0].id)
    assert outcome.opportunity_score == 12.0
    co = repo.list_customer_outcomes(job_id)[0]
    assert co.customer_value.value_metrics[0].opportunity_score == 10.0
```
