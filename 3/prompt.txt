import json
import sqlite3
from datetime import datetime


class IntentStore:

  def __init__(self, db_path="intent_store.db"):
    # Connect to SQLite in WAL mode for high-performance concurrency
    self.conn = sqlite3.connect(db_path)
    self.conn.execute("PRAGMA journal_mode=WAL;")
    self._init_db()

  def _init_db(self):
    """Creates an immutable, append-only event table."""
    with self.conn:
      self.conn.execute("""
                CREATE TABLE IF NOT EXISTS events (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp TEXT NOT NULL,
                    event_type TEXT NOT NULL,
                    payload TEXT NOT NULL
                )
            """)

  def append(self, event_type: str, payload: dict):
    """Appends a new event to the stream (Immutable write)."""
    timestamp = datetime.utcnow().isoformat()
    payload_json = json.dumps(payload)
    with self.conn:
      self.conn.execute(
          "INSERT INTO events (timestamp, event_type, payload) VALUES (?, ?, ?)",
          (timestamp, event_type, payload_json),
      )

  def get_stream(self):
    """Retrieves the full chronological event stream for AI reasoning."""
    cursor = self.conn.cursor()
    cursor.execute(
        "SELECT id, timestamp, event_type, payload FROM events ORDER BY id ASC"
    )
    return [
        {
            "id": row[0],
            "timestamp": row[1],
            "type": row[2],
            "payload": json.loads(row[3]),
        }
        for row in cursor.fetchall()
    ]

if __name__ == "__main__":
  store = IntentStore()

  store.append(
      "CUSTOMER_SIGNAL", {
          "source": "gong_call_104",
          "friction": "Secrets deployment broke staging again",
          "severity": "high",
      }
  )

  # 2. Record an AI agent simulation bet
  store.append(
      "AGENT_SIMULATION", {
          "bet": "Auto-Sync Agent",
          "target_friction": "Secrets deployment broke staging again",
          "simulated_impact": "-90% Time-to-Value",
      }
  )

  # 3. Print the full intent history (what an LLM reads)
  print("--- Product Intent Stream ---")
  for event in store.get_stream():
    print(f"[{event['id']}] {event['type']}: {event['payload']}")
Why This Works Elegantly
Zero Migrations: You never have to alter database tables when your product model changes. New events just carry new JSON schemas in their payload.

Instant AI Context: When an agent needs to evaluate a decision, it simply reads get_stream() to see the exact sequence of human and customer events without parsing flat files or messy folder structures.

True Auditability: History can never be accidentally overwritten, ensuring full traceability from raw customer voice to final product bet. "
